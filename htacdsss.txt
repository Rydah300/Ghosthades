RewriteEngine On

# Remove .php extension from URLs
RewriteCond %{REQUEST_FILENAME}.php -f
RewriteRule ^([^/]+)/?$ $1.php [L]

# Redirect /admin to admin.php
RewriteRule ^admin/?$ admin.php [L]

# Redirect /dashboard to dashboard.php
RewriteRule ^dashboard/?$ dashboard.php [L]

# API routes without .php
RewriteRule ^api/extract/?$ api/extract.php [L]
RewriteRule ^api/batch/?$ api/batch.php [L]
RewriteRule ^api/saved/?$ api/saved.php [L]
RewriteRule ^api/telegram/?$ api/telegram.php [L]
RewriteRule ^api/licenses/?$ api/licenses.php [L]
RewriteRule ^api/users/?$ api/users.php [L]
RewriteRule ^api/stats/?$ api/stats.php [L]

# Auto-add .php for any request without extension
RewriteCond %{REQUEST_FILENAME} !-d
RewriteCond %{REQUEST_FILENAME}\.php -f
RewriteRule ^(.*)$ $1.php [L]

# Block direct access to .php files (optional security)
<FilesMatch "\.php$">
    Order Allow,Deny
    Deny from all
</FilesMatch>

# Allow index.php, dashboard, admin, api
<FilesMatch "^(index|dashboard|admin|api/.*)\.php$">
    Order Allow,Deny
    Allow from all
</FilesMatch>

Options -Indexes
php_flag display_errors off
php_flag expose_php off

<Directory "storage">
    Order Allow,Deny
    Deny from all
</Directory>
